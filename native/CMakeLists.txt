cmake_minimum_required(VERSION 3.16)
project(GM1000Native VERSION 0.6 LANGUAGES CXX)
add_executable(gm1000-native gm1000.cpp)
target_compile_features(gm1000-native PRIVATE cxx_std_17)
target_include_directories(gm1000-native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
find_package(Threads REQUIRED)
target_link_libraries(gm1000-native PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(gm1000-native PRIVATE /O2 /EHsc)
else()
    target_compile_options(gm1000-native PRIVATE -O3 -Wall -Wextra)
endif()
